cmake_minimum_required(VERSION 3.16)
get_filename_component(ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# Reconfigure the embedded workload identity when a developer edits a case.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${ENGINE_ROOT}/src/renderer/raylib_lite_wall_bench.c"
    "${CMAKE_CURRENT_LIST_DIR}/main/core_bench.c"
    "${CMAKE_CURRENT_LIST_DIR}/main/render_preview.c")
option(RENDER_BENCH_HOST "Build standalone host benchmark" OFF)
if(RENDER_BENCH_HOST)
    set(_display_default OFF)
else()
    set(_display_default ON)
endif()
option(RENDER_BENCH_DISPLAY "On-screen wall and core preview (separate from microbenchmark)" ${_display_default})
set(RENDER_BENCH_PREVIEW_FRAMES "0" CACHE STRING "Preview frame limit; 0 runs until reset")
if(NOT RENDER_BENCH_PREVIEW_FRAMES MATCHES "^[0-9]+$")
    message(FATAL_ERROR "RENDER_BENCH_PREVIEW_FRAMES must be non-negative")
endif()
set(RENDER_BENCH_SUITE "wall" CACHE STRING "wall or core")
set(M2D_WALL_MODE "3" CACHE STRING "0 legacy, 1 exact, 2 fixed, 3 error bounded")
set(M2D_WALL_FIXED_PIXELS "16" CACHE STRING "Fixed segment length")
set(M2D_WALL_ERROR_TEXELS "0.25" CACHE STRING "UV error budget")
option(M2D_WALL_AUDIT "UV audit; timings not eligible" OFF)
option(RENDER_BENCH_PIE "Use ESP32-S31 PIE copy/fill" OFF)
option(RENDER_BENCH_LUT_INTERNAL "Place wall LUT in internal RAM" OFF)
if(NOT RENDER_BENCH_SUITE MATCHES "^(wall|core)$")
    message(FATAL_ERROR "RENDER_BENCH_SUITE must be wall or core")
endif()
if(M2D_WALL_AUDIT AND NOT RENDER_BENCH_SUITE STREQUAL "wall")
    message(FATAL_ERROR "UV audit belongs to wall suite")
endif()
if(RENDER_BENCH_DISPLAY AND (M2D_WALL_AUDIT OR NOT RENDER_BENCH_SUITE STREQUAL "wall"))
    message(FATAL_ERROR "Display preview requires wall suite with audit OFF")
endif()
if(RENDER_BENCH_HOST)
    project(render_benchmark C)
    enable_testing()
    include(main/sources.cmake)
    add_executable(render_benchmark ${BENCH_SOURCES})
    target_include_directories(render_benchmark PRIVATE ${BENCH_INCLUDES})
    target_compile_definitions(render_benchmark PRIVATE ${BENCH_DEFINITIONS})
    target_compile_options(render_benchmark PRIVATE -O2 -Wall -Wextra -Werror)
    target_link_libraries(render_benchmark PRIVATE m)
    set_property(TARGET render_benchmark PROPERTY C_STANDARD 11)
    add_test(NAME benchmark COMMAND render_benchmark)
    if(RENDER_BENCH_DISPLAY)
        set(_preview_test_sources ${BENCH_SOURCES})
        list(REMOVE_ITEM _preview_test_sources "${CMAKE_CURRENT_SOURCE_DIR}/main/benchmark_main.c")
        add_executable(render_preview_test tests/test_preview.c ${_preview_test_sources})
        target_include_directories(render_preview_test PRIVATE ${BENCH_INCLUDES})
        target_compile_definitions(render_preview_test PRIVATE ${BENCH_DEFINITIONS})
        target_compile_options(render_preview_test PRIVATE -O2 -Wall -Wextra -Werror)
        target_link_libraries(render_preview_test PRIVATE m)
        add_test(NAME preview_controls_and_pixels COMMAND render_preview_test)
    endif()
else()
    set(COMPONENTS main)
    if(RENDER_BENCH_DISPLAY)
        # Only preview needs the board. Offscreen builds stay BSP-independent.
        if(NOT MOSAICO_BSP_COMPONENT_DIR AND DEFINED ENV{MOSAICO_BSP_COMPONENT_DIR})
            set(MOSAICO_BSP_COMPONENT_DIR "$ENV{MOSAICO_BSP_COMPONENT_DIR}")
        endif()
        if(NOT EXISTS "${MOSAICO_BSP_COMPONENT_DIR}/CMakeLists.txt")
            message(FATAL_ERROR "Display requires -DMOSAICO_BSP_COMPONENT_DIR=/path/to/esp-mosaico-bsp/component")
        endif()
        list(APPEND EXTRA_COMPONENT_DIRS "${MOSAICO_BSP_COMPONENT_DIR}")
        list(APPEND COMPONENTS esp-mosaico-bsp)
    endif()
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
    if(RENDER_BENCH_DISPLAY)
        list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.display.defaults")
    endif()
    if(NOT SDKCONFIG)
        set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Per-build config")
    endif()
    include($ENV{IDF_PATH}/tools/cmake/project.cmake)
    project(render_benchmark)
    if(RENDER_BENCH_DISPLAY)
        # Raw BSP public display.h exposes esp_lcd_touch.h. Without LVGL its
        # private driver dependency does not export that include to consumers.
        idf_component_get_property(_preview_bsp_lib esp-mosaico-bsp COMPONENT_LIB)
        target_link_libraries(${_preview_bsp_lib} PUBLIC idf::espressif__esp_lcd_touch)
    endif()
endif()
